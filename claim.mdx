<Steps>
  <Step title="Send 0.01 SOL to the gateway wallet">

    Wait ~15 seconds after your transaction confirms for the credit to land.
  </Step>
  <Step title="Connect wallet and claim">

(function() {
  var GATEWAY = "https://solana-pulse-gateway-1021990235790.us-central1.run.app";

  }

  function detectProvider() {
    if (window.phantom && window.phantom.solana) return { name: "Phantom", provider: window.phantom.solana };
    if (window.solflare && window.solflare.isSolflare) return { name: "Solflare", provider: window.solflare };
    if (window.backpack) return { name: "Backpack", provider: window.backpack };
    if (window.solana) return { name: "Solana wallet", provider: window.solana };
    return null;
  }

  async function connect() {
    var found = detectProvider();
    if (!found) {
      setStatus('No Solana wallet extension detected. Install <a href="https://phantom.app" target="_blank" style="color:#14F195">Phantom</a> or <a href="https://solflare.com" target="_blank" style="color:#14F195">Solflare</a> and refresh.', "error");

      var infoEl = el("claim-wallet-info");
      infoEl.style.display = "block";
      infoEl.innerHTML = "Connected: " + walletPubkey;
      setStatus("Fetching challenge from gateway...", "info");
      var r = await fetch(GATEWAY + "/api/keys/challenge");

## Troubleshooting

<AccordionGroup>
  <Accordion title="No wallet detected">
    Install [Phantom](https://phantom.app), [Solflare](https://solflare.com/download), or [Backpack](https://backpack.app) as a browser extension. Refresh this page after install.
  </Accordion>
  <Accordion title="Claim returned 404 'no pending key'">
    The gateway didn't see your 0.01 SOL yet. Check your transaction confirmed on-chain, then wait ~30 seconds for the webhook to process it. Retry claim with a fresh challenge (reconnect wallet).
  </Accordion>
  <Accordion title="Claim returned 401 'invalid signature'">
    You signed with a different wallet than the one that sent the SOL. Disconnect, switch active wallet, and reconnect.
  </Accordion>
  <Accordion title="Lost my key">
    Send another 0.01 SOL from the same wallet to generate a new one. Keys are not recoverable.
  </Accordion>
</AccordionGroup>
